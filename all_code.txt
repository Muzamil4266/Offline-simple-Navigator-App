    private fun battery(): Int {
        val i = registerReceiver(null, IntentFilter(Intent.ACTION_BATTERY_CHANGED)) ?: return -1
        val level = i.getIntExtra(BatteryManager.EXTRA_LEVEL, -1)
        val scale = i.getIntExtra(BatteryManager.EXTRA_SCALE, 100)
        return if (level < 0 || scale <= 0) -1 else level * 100 / scale
    }

    private fun refresh() {
        val t = BreadcrumbTracker
        val cur = t.current

        if (t.returnMode) {
            if (t.arrived) {
                tvBig1.text = "HOME"
                tvLabel1.text = "ARRIVED AT START"
            } else {
                tvBig1.text = fmt(t.targetDistance())
                tvLabel1.text = "TO NEXT PIN - ${t.pinsLeft()} LEFT"
            }
            tvBig1.setTextColor(GREEN)
        } else {
            tvBig1.text = fmt(t.distFromLastPin())
            tvLabel1.text = "FROM LAST PIN"
            tvBig1.setTextColor(WHITE)
        }
        tvBig2.text = fmt(t.distFromStart())
        tvLabel2.text = "FROM START - WALKED ${fmt(t.pathDistance())}"

        val weak = hasFine() && cur != null && t.gpsLost
        val gps = when {
            !hasFine() -> "LOCATION PERMISSION NEEDED (TAP HERE)"
            cur == null -> "SEARCHING GPS..."
            weak -> "WEAK GPS - last fix +-${cur.accuracy.roundToInt()} m"
            else -> "GPS +-${cur.accuracy.roundToInt()} m"
        }
        val bat = battery()
        tvStatus.text = "$gps  |  PINS ${t.pins.size}  |  BAT ${if (bat >= 0) "$bat%" else "?"}"
        tvStatus.setTextColor(if (weak || !hasFine()) RED else GRAY)

        tvCoords.text = if (cur != null) {
            String.format(Locale.US, "%.6f, %.6f", cur.latitude, cur.longitude)
        } else "-- , --"

        // return button look
        btnReturn.text = if (t.returnMode) "TRACK" else "RETURN"
        val bg = btnReturn.background as GradientDrawable
        bg.setColor(if (t.returnMode) GREY_BTN else GREEN_BTN)
        btnReturn.setTextColor(if (t.returnMode) WHITE else BLACK)
        btnMode.text = if (canvasView.follow) "SHOW FULL TRAIL" else "FOLLOW ME"

        if (cur != null) {
            declination = GeomagneticField(
                cur.latitude.toFloat(), cur.longitude.toFloat(), cur.altitude.toFloat(),
                System.currentTimeMillis()
            ).declination
        }
        applyHeading()
        canvasView.invalidate()
    }

    // -------------------------------------------------------------- compass

    private fun norm(d: Float): Float = ((d % 360f) + 360f) % 360f

    private fun applyHeading() {
        val cur = BreadcrumbTracker.current
        val h: Float = if (rot != null) {
            norm(compassMag + declination)
        } else if (cur != null && cur.hasBearing() && cur.speed > 0.8f) {
            cur.bearing // no compass: GPS course works only while moving
        } else {
            canvasView.headingDeg
        }
        if (abs(h - lastShownHeading) > 1f) {
            lastShownHeading = h
            canvasView.headingDeg = h
            canvasView.invalidate()
        }
    }

    override fun onSensorChanged(event: SensorEvent) {
        if (event.sensor.type != Sensor.TYPE_ROTATION_VECTOR) return
        val rm = FloatArray(9)
        SensorManager.getRotationMatrixFromVector(rm, event.values)
        // Phone held upright (normal walking pose): heading of the back camera direction.
        val use = if (abs(rm[8]) < 0.7f) {
            val r2 = FloatArray(9)
            SensorManager.remapCoordinateSystem(rm, SensorManager.AXIS_X, SensorManager.AXIS_Z, r2)
            r2
        } else rm
        val o = FloatArray(3)
        SensorManager.getOrientation(use, o)
        val az = norm(Math.toDegrees(o[0].toDouble()).toFloat())
        val delta = ((az - compassMag + 540f) % 360f) - 180f
        compassMag = norm(compassMag + 0.2f * delta)
        applyHeading()
    }

    override fun onAccuracyChanged(sensor: Sensor?, accuracy: Int) {}
}

package com.survival.breadcrumb

import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.app.Service
import android.content.Intent
import android.content.pm.ServiceInfo
import android.graphics.drawable.Icon
import android.location.Location
import android.location.LocationListener
import android.location.LocationManager
import android.os.Build
import android.os.Bundle
import android.os.IBinder
import android.os.Looper
import android.os.SystemClock

/** Keeps GPS + trail alive when screen is off or app is in background. */
class TrackerService : Service(), LocationListener {

    companion object {
        const val ACTION_STOP = "com.survival.breadcrumb.STOP"
        private const val CHANNEL = "trail"
        private const val NOTIF_ID = 1
        private const val FAST_MS = 1000L
        private const val SLOW_MS = 5000L
        private const val STILL_AFTER_MS = 60_000L

        @Volatile
        var running = false
    }

    private lateinit var lm: LocationManager
    private var registered = false
    private var slow = false

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onCreate() {
        super.onCreate()
        BreadcrumbTracker.init(this)
        lm = getSystemService(LOCATION_SERVICE) as LocationManager
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        if (intent?.action == ACTION_STOP) {
            stopSelf()
            return START_NOT_STICKY
        }
        try {
            startInForeground()
        } catch (e: Exception) {
            // Android refused foreground start (background restart rules). Give up cleanly.
            stopSelf()
            return START_NOT_STICKY
        }
        running = true
        if (!registered) {
            registered = request(false)
            if (!registered) {
                stopSelf()
                return START_NOT_STICKY
            }
        }
        return START_STICKY
    }

    override fun onDestroy() {
        try {
            lm.removeUpdates(this)
        } catch (e: SecurityException) {
            // ignore
        }
        registered = false
        running = false
        super.onDestroy()
    }

    // ------------------------------------------------------------------ GPS

    private fun request(slowMode: Boolean): Boolean {
        return try {
            lm.removeUpdates(this)
            lm.requestLocationUpdates(
                LocationManager.GPS_PROVIDER,
                if (slowMode) SLOW_MS else FAST_MS,
                0f,
                this,
                Looper.getMainLooper()
            )
            slow = slowMode
            true
        } catch (e: SecurityException) {
            false
        } catch (e: IllegalArgumentException) {
            false
        }
    }

    override fun onLocationChanged(location: Location) {
        BreadcrumbTracker.onLocation(location)
        // Battery saver: standing still for a minute -> poll slower.
        val still = SystemClock.elapsedRealtime() - BreadcrumbTracker.lastMoveTime > STILL_AFTER_MS
        if (still != slow) request(still)
    }

    // Must be implemented for API < 30.
    @Deprecated("Deprecated in Java")
    override fun onStatusChanged(provider: String?, status: Int, extras: Bundle?) {}

    override fun onProviderEnabled(provider: String) {}

    override fun onProviderDisabled(provider: String) {}

    // --------------------------------------------------------- notification

    @Suppress("DEPRECATION")
    private fun startInForeground() {
        if (Build.VERSION.SDK_INT >= 26) {
            val nm = getSystemService(NotificationManager::class.java)
            if (nm.getNotificationChannel(CHANNEL) == null) {
                nm.createNotificationChannel(
                    NotificationChannel(CHANNEL, "Trail tracking", NotificationManager.IMPORTANCE_LOW)
                )
            }
        }
        val open = PendingIntent.getActivity(
            this, 0, Intent(this, MainActivity::class.java), PendingIntent.FLAG_IMMUTABLE
        )
        val stop = PendingIntent.getService(
            this, 1,
            Intent(this, TrackerService::class.java).setAction(ACTION_STOP),
            PendingIntent.FLAG_IMMUTABLE
        )
        val b = if (Build.VERSION.SDK_INT >= 26) Notification.Builder(this, CHANNEL)
        else Notification.Builder(this)
        val action = Notification.Action.Builder(
            Icon.createWithResource(this, android.R.drawable.ic_menu_close_clear_cancel),
            "STOP",
            stop
        ).build()
        val n = b.setSmallIcon(android.R.drawable.ic_menu_mylocation)
            .setContentTitle("Breadcrumb tracking")
            .setContentText("Recording trail. Tap to open.")
            .setContentIntent(open)
            .setOngoing(true)
            .addAction(action)
            .build()

        if (Build.VERSION.SDK_INT >= 29) {
            startForeground(NOTIF_ID, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_LOCATION)
        } else {
            startForeground(NOTIF_ID, n)
        }
    }
}

package com.survival.breadcrumb

import android.content.Context
import android.location.Location
import android.os.SystemClock
import java.io.File
import java.io.IOException

data class Pin(val lat: Double, val lon: Double, val manual: Boolean, val time: Long)

/**
 * Pure logic + shared state. Lives in the process, used by TrackerService (writes)
 * and MainActivity (reads). Everything runs on the main thread.
 */
object BreadcrumbTracker {
    const val AUTO_PIN_M = 30f
    private const val GOOD_ACC = 18f          // normal accuracy gate (m)
    private const val WEAK_ACC = 35f          // relaxed gate after 30 s without good fix
    private const val WEAK_AFTER_MS = 30_000L
    private const val REACH_M = 10f           // pin counts as reached inside this radius
    private const val MAX_PINS = 4000

    val pins = ArrayList<Pin>()

    var current: Location? = null
        private set
    var returnMode = false
        private set
    var returnIdx = -1
        private set
    var arrived = false
        private set
    var lastMoveTime = SystemClock.elapsedRealtime()
        private set

    /** UI hook. Set in onResume, cleared in onPause. */
    var onChange: (() -> Unit)? = null

    private var file: File? = null
    private var startedAt = SystemClock.elapsedRealtime()
    private var lastAcceptedAt = 0L
    private var lastMovePos: Location? = null
    private val out = FloatArray(1)
    private val tmp = Location("target")

    val gpsLost: Boolean
        get() = SystemClock.elapsedRealtime() - maxOf(lastAcceptedAt, startedAt) > WEAK_AFTER_MS

    // ---------------------------------------------------------------- setup

    fun init(ctx: Context) {
        if (file != null) return
        file = File(ctx.applicationContext.filesDir, "trail.csv")
        startedAt = SystemClock.elapsedRealtime()
        load()
    }

    // ------------------------------------------------------------ GPS input

    fun onLocation(loc: Location) {
        val now = SystemClock.elapsedRealtime()
        val sinceGood = now - maxOf(lastAcceptedAt, startedAt)
        val limit = if (sinceGood < WEAK_AFTER_MS) GOOD_ACC else WEAK_ACC
        if (!loc.hasAccuracy() || loc.accuracy > limit) return

        lastAcceptedAt = now
        current = loc

        val mp = lastMovePos
        if (mp == null || loc.distanceTo(mp) > maxOf(5f, loc.accuracy)) {
            lastMovePos = loc
            lastMoveTime = now
        }

        if (pins.isEmpty()) {
            addPin(loc, false) // START
        } else if (!returnMode) {
            val last = pins.last()
            val d = dist(loc.latitude, loc.longitude, last.lat, last.lon)
            // d > accuracy: standing still with drift must not make pins
            if (d >= AUTO_PIN_M && d > loc.accuracy) addPin(loc, false)
        } else {
            updateReturn(loc)
        }
        onChange?.invoke()
    }

    // ------------------------------------------------------------- actions

    fun dropManual(): Boolean {
        val c = current ?: return false
        if (returnMode) return false
        addPin(c, true)
        onChange?.invoke()
        return true
    }

    fun startReturn() {
        if (pins.isEmpty()) return
        returnMode = true
        arrived = false
        returnIdx = pins.lastIndex
        current?.let { updateReturn(it) }
        onChange?.invoke()
    }

    fun stopReturn() {
        returnMode = false
        arrived = false
        onChange?.invoke()
    }

    fun reset() {
        pins.clear()
        returnMode = false
        arrived = false
        returnIdx = -1
        lastMovePos = null
        try {
            file?.delete()
        } catch (e: SecurityException) {
            // ignore
        }
        onChange?.invoke()
    }

    // ------------------------------------------------------------- queries

    fun distFromLastPin(): Float? {
        val c = current ?: return null
        val p = pins.lastOrNull() ?: return null
        return dist(c.latitude, c.longitude, p.lat, p.lon)
    }

    fun distFromStart(): Float? {
        val c = current ?: return null
        val p = pins.firstOrNull() ?: return null
        return dist(c.latitude, c.longitude, p.lat, p.lon)
    }

    /** Total walked length along pins (+ live leg). */
    fun pathDistance(): Float {
        var sum = 0f
        for (i in 1 until pins.size) {
            sum += dist(pins[i - 1].lat, pins[i - 1].lon, pins[i].lat, pins[i].lon)
        }
        val c = current
        val last = pins.lastOrNull()
        if (c != null && last != null) sum += dist(c.latitude, c.longitude, last.lat, last.lon)
        return sum
    }

    fun target(): Pin? =
        if (returnMode && !arrived && returnIdx in pins.indices) pins[returnIdx] else null

    fun pinsLeft(): Int = if (returnMode && !arrived) returnIdx + 1 else 0

    fun targetDistance(): Float? {
        val c = current ?: return null
        val t = target() ?: return null
        return dist(c.latitude, c.longitude, t.lat, t.lon)
    }

    /** Bearing to next return pin, degrees clockwise from true north. */
    fun targetBearing(): Float? {
        val c = current ?: return null
        val t = target() ?: return null
        tmp.latitude = t.lat
        tmp.longitude = t.lon
        return c.bearingTo(tmp)
    }

    // ------------------------------------------------------------ internals

    private fun updateReturn(loc: Location) {
        if (arrived) return
        val reach = maxOf(REACH_M, loc.accuracy)
        while (returnIdx >= 0) {
            val p = pins[returnIdx]
            val d = dist(loc.latitude, loc.longitude, p.lat, p.lon)
            if (d <= reach) {
                returnIdx--
                continue
            }
            // Cut corner: already closer to the pin before this one -> skip this one.
            if (returnIdx > 0) {
                val prev = pins[returnIdx - 1]
                if (dist(loc.latitude, loc.longitude, prev.lat, prev.lon) < d) {
                    returnIdx--
                    continue
                }
            }
            break
        }
        if (returnIdx < 0) {
            arrived = true
            returnIdx = 0
        }
    }

    private fun addPin(loc: Location, manual: Boolean) {
        val p = Pin(loc.latitude, loc.longitude, manual, System.currentTimeMillis())
        pins.add(p)
        try {
            file?.appendText(line(p))
        } catch (e: IOException) {
            // storage trouble: keep tracking in memory
        }
        if (pins.size > MAX_PINS) thin()
    }

    /** Memory cap: drop every second auto pin. Keeps START, last pin, manual pins. */
    private fun thin() {
        val keep = ArrayList<Pin>()
        var skip = false
        for ((i, p) in pins.withIndex()) {
            if (i == 0 || i == pins.lastIndex || p.manual) {
                keep.add(p)
            } else {
                if (!skip) keep.add(p)
                skip = !skip
            }
        }
        pins.clear()
        pins.addAll(keep)
        if (returnIdx > pins.lastIndex) returnIdx = pins.lastIndex
        try {
            file?.writeText(pins.joinToString("") { line(it) })
        } catch (e: IOException) {
            // ignore
        }
    }

    private fun line(p: Pin) = "${p.lat},${p.lon},${if (p.manual) 1 else 0},${p.time}\n"

    private fun load() {
        val f = file ?: return
        if (!f.exists()) return
        try {
            f.forEachLine { l ->
                val s = l.split(',')
                if (s.size >= 4) {
                    try {
                        pins.add(Pin(s[0].toDouble(), s[1].toDouble(), s[2] == "1", s[3].toLong()))
                    } catch (e: NumberFormatException) {
                        // skip bad line
                    }
                }
            }
        } catch (e: IOException) {
            // ignore
        }
    }

    private fun dist(lat1: Double, lon1: Double, lat2: Double, lon2: Double): Float {
        Location.distanceBetween(lat1, lon1, lat2, lon2, out)
        return out[0]
    }
}
